- __Item #:__ npc-controls
- __Tags:__ ui, actuation
- __Activation:__ code, keyboard, pointer

```sh
# pointer moves player
# @bootable
pick meta.{floor,do} | move $( playerPath ) --force

# keys move player
# @bootable
wasd_delta --nav $( playerPath ) --force | move $( playerPath ) --force

# last picked moves
pick meta.{floor,do} | move $( lastPicked ) --force

# pointer moves named npc
pick meta.{floor,do} | move npc:rob --backstep --force

# keys move named npc
wasd_delta --nav npc:rob | move npc:rob --force

# pointer strafes npc
pick meta.{floor,do} | move rob --strafe --force

# keys strafe npc
wasd_delta npc:rob | move npc:rob --strafe --force

# long pressed looks
pick --long | look npc:rob --force

# keys look
wasd_delta npc:rob | look npc:rob --force
```
