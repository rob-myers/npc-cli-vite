```sh title="control player or last picked"
# pointer moves player
# @bootable
pick meta.{floor,do} | move $( playerPath ) --force

# keys wasd move player
# @bootable @default=desktop
wasd_delta --nav $( playerPath ) --force | move $( playerPath ) --force

# pointer moves last picked
# @bootable @default
pick meta.{floor,do} | move $( lastPicked ) --force

# keys wasd move last picked
# @bootable
wasd_delta --nav $( lastPicked ) --force | move $( lastPicked ) --force
```

```sh title="control rob" rob=npc
# rob moved by pointer
# @bootable
pick meta.{floor,do} | move rob --force

# rob strafed by pointer
pick meta.{floor,do} | move rob --strafe --force

# rob looks via long pointer
pick --long | look rob --force

# rob moved by wasd keys
wasd_delta --nav rob | move rob --force

# rob strafed by wasd keys
wasd_delta rob | move rob --strafe --force

# rob looks via wasd keys
wasd_delta rob | look rob --force
```
